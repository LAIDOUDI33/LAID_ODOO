   424→   424→          internalNotes: editingOrder.internalNotes || '',
   425→   425→          supplierNotes: editingOrder.supplierNotes || '',
   426→   426→          lines:
   427→   427→            editingOrder.lines.length > 0
   428→   428→              ? editingOrder.lines.map((l) => ({
   429→   429→                  id: l.id,
   430→   430→                  productId: l.productId,
   431→   431→                  description: l.description,
   432→   432→                  quantity: l.quantity,
   433→   433→                  unitPrice: l.unitPrice,
   434→   434→                  discountRate: l.discountRate,
   435→   435→                  tvaRate: l.tvaRate,
   436→   436→                }))
   437→   437→              : [{ productId: '', quantity: 1, unitPrice: 0, tvaRate: 19 }],
   438→   438→        })
   439→   439→      } else {
   440→   440→        setFormData({
   441→   441→          partnerId: '',
   442→   442→          expectedDate: '',
   443→   443→          paymentTerms: '30',
   444→   444→          incoterm: '',
   445→   445→          internalNotes: '',
   446→   446→          supplierNotes: '',

   453→   453→  async function loadPartners() {

   479→   479→  function updateField(field: keyof CreatePOFormData, value: string) {

   481→   481→  }

   483→   483→  function updateLine(index: number, field: keyof PurchaseOrderLine, value: any) {
   484→   484→    setFormData((prev) => ({
   485→   485→      ...prev,
   486→   486→      lines: prev.lines.map((line, i) =>
   487→   487→        i === index ? { ...line, [field]: value } : line
   488→   488→      ),
   489→   489→    }))
   490→   490→  }
   491→   491→
   492→   492→  function addLine() {
   493→   493→    setFormData((prev) => ({
   494→   494→      ...prev,
   495→   495→      lines: [
   496→   496→        ...prev.lines,
   497→   497→        { productId: '', quantity: 1, unitPrice: 0, tvaRate: 19 },
   498→   498→      ],
   499→   499→    }))
   500→   500→  }
   501→   501→
   502→   502→  function removeLine(index: number) {
   503→   503→    setFormData((prev) => ({
   504→   504→      ...prev,
   505→   505→      lines: prev.lines.filter((_, i) => i !== index),
   506→   506→    }))
   507→   507→  }
   508→   508→
   509→   509→  // Calculate totals
   510→   510→  const totals = useMemo(() => {
   511→   511→    let amountUntaxed = 0
   512→   512→    let amountTax = 0
   513→   513→
   514→   514→    for (const line of formData.lines) {
   515→   515→      const qty = line.quantity || 0
   516→   516→      const price = line.unitPrice || 0
   517→   517→      const discount = line.discountRate || 0
   518→   518→      const tvaRate = line.tvaRate ?? 19
   519→   519→
   520→   520→      const lineTotal = qty * price * (1 - discount / 100)
   521→   521→      amountUntaxed += lineTotal
   522→   522→      amountTax += lineTotal * (tvaRate / 100)
   523→   523→    }
   524→   524→
   525→   525→    return {
   526→   526→      amountUntaxed: Math.round(amountUntaxed * 100) / 100,
   527→   527→      amountTax: Math.round(amountTax * 100) / 100,
   528→   528→      amountTotal: Math.round((amountUntaxed + amountTax) * 100) / 100,
   529→   529→    }
   530→   530→  }, [formData.lines])
