     6→     6→const VALID_STATUSES = ['draft', 'sent', 'confirmed', 'processing', 'delivered', 'invoiced', 'done', 'cancelled'];
     7→     7→
     8→     8→// Status transition rules
     9→     9→const VALID_TRANSITIONS: Record<string, string[]> = {
    10→    10→  draft: ['sent', 'cancelled'],
    11→    11→  sent: ['confirmed', 'draft', 'cancelled'],
    12→    12→  confirmed: ['processing', 'sent', 'cancelled'],
    13→    13→  processing: ['delivered', 'confirmed'],
    14→    14→  delivered: ['invoiced', 'processing'],
    15→    15→  invoiced: ['done', 'delivered'],
    16→    16→  done: [],
    17→    17→  cancelled: ['draft'] // Can reinstate cancelled orders
    18→    18→};

    20→    20→// ============================================================
    21→    21→// Helper: Validate status transition
    22→    22→// ============================================================
    23→    23→function canTransition(currentStatus: string, newStatus: string): boolean {
    24→    24→  if (!VALID_TRANSITIONS[currentStatus]) return false;
    25→    25→  return VALID_TRANSITIONS[currentStatus].includes(newStatus);
    26→    26→}

   114→   114→// PUT /api/sales-orders/[id] - Update a sales order
   115→   115→// ============================================================
   116→   116→export async function PUT(
   117→   117→  request: Request,
   118→   118→  { params }: { params: Promise<{ id: string }> }
   119→   119→) {
   120→   120→  try {
   121→   121→    const { id } = await params;
   122→   122→    const body = await request.json();
   123→   123→
   124→   124→    // Check if sales order exists
   125→   125→    const existingOrder = await db.salesOrder.findUnique({
   126→   126→      where: { id },
   127→   127→      include: { lines: true }
   128→   128→    });
   129→   129→
   130→   130→    if (!existingOrder) {
   131→   131→      return NextResponse.json(
   132→   132→        { success: false, error: 'Sales order not found' },
   133→   133→        { status: 404 }
   134→   134→      );
   135→   135→    }
   136→   136→
   137→   137→    // Check if order can be modified (only draft/sent can be edited)
   138→   138→    if (!['draft', 'sent'].includes(existingOrder.status)) {
   139→   139→      return NextResponse.json(
   140→   140→        { success: false, error: `Cannot modify sales order with status '${existingOrder.status}'. Only draft or sent orders can be modified.` },
   141→   141→        { status: 400 }
   142→   142→      );
   143→   143→    }
   144→   144→
   145→   145→    const {
   146→   146→      expectedDate,
   147→   147→      paymentTerms,
   148→   148→      paymentMode,
   149→   149→      shippingAddress,
   150→   150→      warehouseId,
   151→   151→      internalNotes,
   152→   152→      customerNotes,
   153→   153→      lines,
   154→   154→      status
   155→   155→    } = body;
   156→   156→
   157→   157→    // Validate status change if provided
   158→   158→    if (status && status !== existingOrder.status) {
   159→   159→      if (!VALID_STATUSES.includes(status)) {
   160→   160→        return NextResponse.json(
   161→   161→          { success: false, error: `Invalid status. Must be one of: ${VALID_STATUSES.join(', ')}` },
   162→   162→          { status: 400 }
   163→   163→        );
   164→   164→      }
   165→   165→      if (!canTransition(existingOrder.status, status)) {
   166→   166→        return NextResponse.json(
   167→   167→          { success: false, error: `Cannot transition from '${existingOrder.status}' to '${status}'` },
   168→   168→          { status: 400 }
   169→   169→        );
   170→   170→      }
   171→   171→    }
   172→   172→
   173→   173→    // Check warehouse if provided
   174→   174→    if (warehouseId && warehouseId !== existingOrder.warehouseId) {
   175→   175→      const warehouse = await db.warehouse.findUnique({ where: { id: warehouseId } });
   176→   176→      if (!warehouse) {
   177→   177→        return NextResponse.json(
   178→   178→          { success: false, error: 'Warehouse not found' },
   179→   179→          { status: 404 }
   180→   180→        );
   181→   181→      }
   182→   182→    }
   183→   183→
   184→   184→    // Prepare update data
   185→   185→    const updateData: any = {};
   186→   186→    
   187→   187→    if (expectedDate !== undefined) updateData.expectedDate = expectedDate ? new Date(expectedDate) : null;
   188→   188→    if (paymentTerms !== undefined) updateData.paymentTerms = paymentTerms;
   189→   189→    if (paymentMode !== undefined) updateData.paymentMode = paymentMode;
   190→   190→    if (shippingAddress !== undefined) updateData.shippingAddress = shippingAddress;
   191→   191→    if (warehouseId !== undefined) updateData.warehouseId = warehouseId;
   192→   192→    if (internalNotes !== undefined) updateData.internalNotes = internalNotes;
   193→   193→    if (customerNotes !== undefined) updateData.customerNotes = customerNotes;
   194→   194→    if (status !== undefined) updateData.status = status;
   195→   195→
   196→   196→    // Handle line updates if provided
   197→   197→    let calculatedAmounts = null;
   198→   198→    
   199→   199→    if (lines && Array.isArray(lines)) {
   200→   200→      if (lines.length === 0) {
   201→   201→        return NextResponse.json(
   202→   202→          { success: false, error: 'At least one line item is required' },
   203→   203→          { status: 400 }
   204→   204→        );
   205→   205→      }
   206→   206→
   207→   207→      // Validate new lines
   208→   208→      for (let i = 0; i < lines.length; i++) {
   209→   209→        const line = lines[i];
   210→   210→        if (!line.productId) {
   211→   211→          return NextResponse.json(
   212→   212→            { success: false, error: `Line ${i + 1}: Product is required` },
   213→   213→            { status: 400 }
   214→   214→          );
   215→   215→        }
   216→   216→        if (!line.quantity || parseFloat(line.quantity) <= 0) {
   217→   217→          return NextResponse.json(
   218→   218→            { success: false, error: `Line ${i + 1}: Quantity must be greater than 0` },
   219→   219→            { status: 400 }
   220→   220→          );
   221→   221→        }
   222→   222→      }
   223→   223→
   224→   224→      // Process new lines with calculations
   225→   225→      const processedLines = lines.map((line: any) => {
   226→   226→        const quantity = parseFloat(line.quantity) || 0;
   227→   227→        const unitPrice = parseFloat(line.unitPrice) || 0;
   228→   228→        const discountRate = parseFloat(line.discountRate) || 0;
   229→   229→        let tvaRate = parseFloat(line.tvaRate);
   230→   230→        
   231→   231→        if (isNaN(tvaRate)) tvaRate = 0.19;
   232→   232→
   233→   233→        const amountUntaxedBeforeDiscount = quantity * unitPrice;
   234→   234→        const amountUntaxed = Math.round(amountUntaxedBeforeDiscount * (1 - discountRate / 100) * 100) / 100;
   235→   235→        const amountTax = Math.round(amountUntaxed * tvaRate * 100) / 100;
   236→   236→        const amountTotal = amountUntaxed + amountTax;
   237→   237→
   238→   238→        return {
   239→   239→          productId: line.productId,
   240→   240→          description: line.description || null,
   241→   241→          quantity,
   242→   242→          unitPrice,
   243→   243→          discountRate,
   244→   244→          tvaRate,
   245→   245→          amountUntaxed,
   246→   246→          amountTax,
   247→   247→          amountTotal
   248→   248→        };
   249→   249→      });
   250→   250→
   251→   251→      // Calculate totals
   252→   252→      const tvaResult = calculateTVACollectee(processedLines.map((l) => ({ 
   253→   253→        amountUntaxed: l.amountUntaxed, 
   254→   254→        tvaRate: l.tvaRate 
   255→   255→      })));
   256→   256→
   257→   257→      const timbreFiscal = getTimbreFiscal('facture', tvaResult.totalTTC);
   258→   258→
   259→   259→      calculatedAmounts = {
   260→   260→        amountUntaxed: tvaResult.totalHT,
   261→   261→        amountTax: tvaResult.totalTVACollectee,
   262→   262→        timbreFiscal,
   263→   263→        amountTotal: tvaResult.totalTTC + timbreFiscal
   264→   264→      };
   265→   265→
   266→   266→      updateData.amountUntaxed = calculatedAmounts.amountUntaxed;
   267→   267→      updateData.amountTax = calculatedAmounts.amountTax;
   268→   268→      updateData.timbreFiscal = calculatedAmounts.timbreFiscal;
   269→   269→      updateData.amountTotal = calculatedAmounts.amountTotal;
   270→   270→    }
   271→   271→
   272→   272→    // Update in transaction to handle lines properly
   273→   273→    const updatedOrder = await db.$transaction(async (tx) => {
   274→   274→      // Delete old lines and create new ones if lines are being updated
   275→   275→      if (lines && Array.isArray(lines)) {
   276→   276→        await tx.salesOrderLine.deleteMany({
   277→   277→          where: { salesOrderId: id }
   278→   278→        });
   279→   279→      }
   280→   280→
   281→   281→      const order = await tx.salesOrder.update({
   282→   282→        where: { id },
   283→   283→        data: {
   284→   284→          ...updateData,
   285→   285→          ...(lines && Array.isArray(lines) ? {
   286→   286→            lines: {
   287→   287→              create: lines.map((line: any) => {
   288→   288→                const quantity = parseFloat(line.quantity) || 0;
   289→   289→                const unitPrice = parseFloat(line.unitPrice) || 0;
   290→   290→                const discountRate = parseFloat(line.discountRate) || 0;
   291→   291→                let tvaRate = parseFloat(line.tvaRate);
   292→   292→                if (isNaN(tvaRate)) tvaRate = 0.19;
   293→   293→
   294→   294→                const amountUntaxedBeforeDiscount = quantity * unitPrice;
   295→   295→                const amountUntaxed = Math.round(amountUntaxedBeforeDiscount * (1 - discountRate / 100) * 100) / 100;
   296→   296→                const amountTax = Math.round(amountUntaxed * tvaRate * 100) / 100;
   297→   297→                const amountTotal = amountUntaxed + amountTax;
   298→   298→
   299→   299→                return {
   300→   300→                  productId: line.productId,
   301→   301→                  description: line.description || null,
   302→   302→                  quantity,
   303→   303→                  unitPrice,
   304→   304→                  discountRate,
   305→   305→                  tvaRate,
   306→   306→                  amountUntaxed,
   307→   307→                  amountTax,
   308→   308→                  amountTotal,
   309→   309→                  quantityDelivered: 0,
   310→   310→                  quantityInvoiced: 0
   311→   311→                };
   312→   312→              })
   313→   313→            }
   314→   314→          } : {})
   315→   315→        },
   316→   316→        include: {
   317→   317→          partner: { select: { id: true, name: true } },
   318→   318→          lines: { include: { product: { select: { id: true, name: true } } } }
   319→   319→        }
   320→   320→      });
   321→   321→
   322→   322→      return order;
   323→   323→    });
   324→   324→
   325→   325→    return NextResponse.json({
   326→   326→      success: true,
   327→   327→      data: updatedOrder,
   328→   328→      message: `Sales Order ${updatedOrder.reference} updated successfully`
   329→   329→    });
   330→   330→  } catch (error) {
   331→   331→    console.error('SalesOrders [id] PUT Error:', error);
   332→   332→    return NextResponse.json(
   333→   333→      { success: false, error: 'Failed to update sales order' },
   334→   334→      { status: 500 }
   335→   335→    );
   336→   336→  }
   337→   337→}

   389→   389→    // Cancel the order (soft delete by changing status)

   519→   519→      { status: 400 }
   520→   520→    );
   521→   521→  }
   522→   522→
   523→   523→  // Determine target warehouse
   524→   524→  const targetWarehouseId = warehouseId || order.warehouseId;
   525→   525→  if (!targetWarehouseId) {
   526→   526→    return NextResponse.json(
   527→   527→      { success: false, error: 'No warehouse specified for delivery' },
   528→   528→      { status: 400 }
   529→   529→    );
   530→   530→  }
   531→   531→
   532→   532→  // Verify warehouse exists
   533→   533→  const warehouse = await db.warehouse.findUnique({ where: { id: targetWarehouseId } });
   534→   534→  if (!warehouse) {
   535→   535→    return NextResponse.json(
   536→   536→      { success: false, error: 'Warehouse not found' },
   537→   537→      { status: 404 }
   538→   538→    );
   539→   539→  }
   540→   540→
   541→   541→  // Process delivery in transaction
   542→   542→  const result = await db.$transaction(async (tx) => {
   543→   543→    const movements: any[] = [];
   544→   544→    let totalDelivered = 0;
   545→   545→    const now = new Date();
   546→   546→    const deliveryDt = deliveryDate ? new Date(deliveryDate) : now;
   547→   547→
   548→   548→    // Generate movement reference
   549→   549→    const year = now.getFullYear();
   550→   550→    const month = String(now.getMonth() + 1).padStart(2, '0');
   551→   551→    const moveCount = await tx.stockMovement.count({
   552→   552→      where: {
   553→   553→        reference: { startsWith: `SORT-${year}-${month}` }
   554→   554→      }
   555→   555→    });
   556→   556→    const sequence = String(moveCount + 1).padStart(3, '0');
   557→   557→
   558→   558→    for (const item of items) {
   559→   559→      const line = order.lines.find(l => l.id === item.lineId);
   560→   560→      if (!line) {
   561→   561→        throw new Error(`Line item ${item.lineId} not found in sales order`);
   562→   562→      }
   563→   563→
   564→   564→      const qtyToDeliver = parseFloat(item.quantity);
   565→   565→      if (qtyToDeliver <= 0) {
   566→   566→        throw new Error(`Invalid quantity for line ${item.lineId}`);
   567→   567→      }
   568→   568→
   569→   569→      const remainingQty = line.quantity - line.quantityDelivered;
   570→   570→      if (qtyToDeliver > remainingQty) {
   571→   571→        throw new Error(`Cannot deliver ${qtyToDeliver} units. Only ${remainingQty} remaining for this line.`);
   572→   572→      }
   573→   573→
   574→   574→      // Create stock movement (outgoing)
   575→   575→      const movementRef = `SORT-${year}-${month}-${sequence}-${movements.length + 1}`;
   576→   576→      const movement = await tx.stockMovement.create({
   577→   577→        data: {
   578→   578→          reference: movementRef,
   579→   579→          date: deliveryDt,
   580→   580→          type: 'out',
   581→   581→          quantity: qtyToDeliver,
   582→   582→          unitCost: line.unitPrice, // Use selling price as cost reference
   583→   583→          totalCost: Math.round(qtyToDeliver * line.unitPrice * 100) / 100,
   584→   584→          notes: notes || `Delivery for SO: ${order.reference}`,
   585→   585→          
   586→   586→          productId: line.productId,
   587→   587→          warehouseId: targetWarehouseId,
   588→   588→          
   589→   589→          sourceDoc: 'sales_order',
   590→   590→          sourceId: order.id
   591→   591→        }
   592→   592→      });
   593→   593→      movements.push(movement);
   594→   594→
   595→   595→      // Update line delivered quantity
   596→   596→      const newQtyDelivered = line.quantityDelivered + qtyToDeliver;
   597→   597→      await tx.salesOrderLine.update({
   598→   598→        where: { id: line.id },
   599→   599→        data: { quantityDelivered: newQtyDelivered }
   600→   600→      });
   601→   601→
   602→   602→      // Calculate delivered amount for this line
   603→   603→      const lineRatio = qtyToDeliver / line.quantity;
   604→   604→      totalDelivered += line.amountTotal * lineRatio;
   605→   605→    }
   606→   606→
   607→   607→    // Update order amounts and status
   608→   608→    const newAmountDelivered = order.amountDelivered + totalDelivered;
   609→   609→    let newStatus = order.status;
   610→   610→    
   611→   611→    // Check if fully delivered
   612→   612→    const allLinesFullyDelivered = order.lines.every(line => {
   613→   613→      const deliveredItem = items.find((item: any) => item.lineId === line.id);
   614→   614→      if (deliveredItem) {
   615→   615→        return (line.quantityDelivered + parseFloat(deliveredItem.quantity)) >= line.quantity;
   616→   616→      }
   617→   617→      return line.quantityDelivered >= line.quantity;
   618→   618→    });
   619→   619→
   620→   620→    if (allLinesFullyDelivered && Math.abs(newAmountDelivered - order.amountTotal) < 0.01) {
   621→   621→      newStatus = 'delivered';
   622→   622→    } else if (newAmountDelivered > 0) {
   623→   623→      newStatus = 'processing';
   624→   624→    }
   625→   625→
   626→   626→    const updatedOrder = await tx.salesOrder.update({
   627→   627→      where: { id },
   628→   628→      data: {
   629→   629→        amountDelivered: Math.round(newAmountDelivered * 100) / 100,
   630→   630→        deliveryDate: deliveryDt,
   631→   631→        status: newStatus
   632→   632→      },
   633→   633→      include: {
   634→   634→        partner: { select: { id: true, name: true } },
   635→   635→        lines: { include: { product: { select: { id: true, name: true } } } }
   636→   636→      }
   637→   637→    });
   638→   638→
   639→   639→    return { order: updatedOrder, movements };
   640→   640→  });
   641→   641→
   642→   642→  return NextResponse.json({
   643→   643→    success: true,
   644→   644→    data: result.order,
   645→   645→    movements: result.movements,
   646→   646→    message: `Delivery processed for Sales Order ${result.order.reference}. ${result.movements.length} stock movement(s) created.`
   647→   647→  });
   648→   648→}

   651→   651→// Action: Create Invoice from Sales Order
   652→   652→// ============================================================
   653→   653→async function handleCreateInvoice(id: string, body: any) {
   654→   654→  const order = await db.salesOrder.findUnique({
   655→   655→    where: { id },
   656→   656→    include: {
   657→   657→      lines: { include: { product: true } },
   658→   658→      partner: true,
   659→   659→      company: true
   660→   660→    }
   661→   661→  });
   662→   662→
   663→   663→  if (!order) {
   664→   664→    return NextResponse.json(
   665→   665→      { success: false, error: 'Sales order not found' },
   666→   666→      { status: 404 }
   667→   667→    );
   668→   668→  }
   669→   669→
   670→   670→  // Allow invoicing from confirmed/processing/delivered statuses
   671→   671→  if (!['confirmed', 'processing', 'delivered'].includes(order.status)) {
   672→   672→    return NextResponse.json(
   673→   673→      { success: false, error: `Cannot invoice sales order with status '${order.status}'. Order must be at least confirmed.` },
   674→   674→      { status: 400 }
   675→   675→    );
   676→   676→  }
   677→   677→
   678→   678→  // Get invoice options from body
   679→   679→  const { invoiceAll, items, dueDate, paymentTerms, paymentMode, internalNotes, customerNotes } = body;
   680→   680→  
   681→   681→  // Default to invoicing all if not specified
   682→   682→  const shouldInvoiceAll = invoiceAll !== false;
   683→   683→
   684→   684→  // Determine which lines to invoice
   685→   685→  let linesToInvoice: typeof order.lines;
   686→   686→  
   687→   687→  if (shouldInvoiceAll) {
   688→   688→    linesToInvoice = order.lines.filter(line => line.quantityInvoiced < line.quantity);
   689→   689→  } else if (items && Array.isArray(items)) {
   690→   690→    linesToInvoice = order.lines.filter(line => 
   691→   691→      items.some((item: any) => item.lineId === line.id)
   692→   692→    );
   693→   693→  } else {
   694→   694→    linesToInvoice = order.lines.filter(line => line.quantityInvoiced < line.quantity);
   695→   695→  }
   696→   696→
   697→   697→  if (linesToInvoice.length === 0) {
   698→   698→    return NextResponse.json(
   699→   699→      { success: false, error: 'No eligible lines to invoice (all quantities already invoiced)' },
   700→   700→      { status: 400 }
   701→   701→    );
   702→   702→  }
   703→   703→
   704→   704→  // Create invoice in transaction
   705→   705→  const result = await db.$transaction(async (tx) => {
   706→   706→    const now = new Date();
   707→   707→    
   708→   708→    // Generate invoice reference
   709→   709→    const year = now.getFullYear();
   710→   710→    const month = String(now.getMonth() + 1).padStart(2, '0');
   711→   711→    const invoiceCount = await tx.invoice.count({
   712→   712→      where: {
   713→   713→        companyId: order.companyId,
   714→   714→        reference: { startsWith: `FACT-${year}-${month}` }
   715→   715→      }
   716→   716→    });
   717→   717→    const sequence = String(invoiceCount + 1).padStart(3, '0');
   718→   718→    const invoiceRef = `FACT-${year}-${month}-${sequence}`;
   719→   719→
   720→   720→    // Prepare invoice lines
   721→   721→    const invoiceLines = linesToInvoice.map(line => {
   722→   722→      // Calculate remaining quantity to invoice
   723→   723→      const remainingQty = line.quantity - line.quantityInvoiced;
   724→   724→      
   725→   725→      // If specific quantity requested
   726→   726→      const requestedItem = items?.find((item: any) => item.lineId === line.id);
   727→   727→      const qtyToInvoice = requestedItem 
   728→   728→        ? Math.min(parseFloat(requestedItem.quantity), remainingQty)
   729→   729→        : remainingQty;
   730→   730→
   731→   731→      const lineRatio = qtyToInvoice / line.quantity;
   732→   732→      
   733→   733→      return {
   734→   734→        productId: line.productId,
   735→   735→        label: line.description || line.product?.name || null,
   736→   736→        quantity: qtyToInvoice,
   737→   737→        unitPrice: line.unitPrice,
   738→   738→        discountRate: line.discountRate,
   739→   739→        tvaRate: line.tvaRate,
   740→   740→        amountUntaxed: Math.round(line.amountUntaxed * lineRatio * 100) / 100,
   741→   741→        amountTax: Math.round(line.amountTax * lineRatio * 100) / 100,
   742→   742→        amountTotal: Math.round(line.amountTotal * lineRatio * 100) / 100
   743→   743→      };
   744→   744→    });
   745→   745→
   746→   746→    // Calculate totals
   747→   747→    const tvaResult = calculateTVACollectee(invoiceLines.map(l => ({
   748→   748→      amountUntaxed: l.amountUntaxed,
   749→   749→      tvaRate: l.tvaRate
   750→   750→    })));
   751→   751→
   752→   752→    const timbreFiscal = getTimbreFiscal('facture', tvaResult.totalTTC);
   753→   753→    const amountUntaxed = tvaResult.totalHT;
   754→   754→    const amountTax = tvaResult.totalTVACollectee;
   755→   755→    const amountTotal = tvaResult.totalTTC + timbreFiscal;
   756→   756→
   757→   757→    // Set due date
   758→   758→    const invoiceDueDate = dueDate 
   759→   759→      ? new Date(dueDate) 
   760→   760→      : new Date(now.getTime() + (parseInt(paymentTerms || order.paymentTerms || '30') * 24 * 60 * 60 * 1000));
   761→   761→
   762→   762→    // Create invoice
   763→   763→    const invoice = await tx.invoice.create({
   764→   764→      data: {
   765→   765→        reference: invoiceRef,
   766→   766→        date: now,
   767→   767→        dueDate: invoiceDueDate,
   768→   768→        status: 'draft',
   769→   769→        type: 'invoice',
   770→   770→        
   771→   771→        amountUntaxed,
   772→   772→        amountTax,
   773→   773→        timbreFiscal,
   774→   774→        amountTotal,
   775→   775→        amountPaid: 0,
   776→   776→        amountDue: amountTotal,
   777→   777→        
   778→   778→        partnerId: order.partnerId,
   779→   779→        companyId: order.companyId,
   780→   780→        salesOrderId: order.id,
   781→   781→        
   782→   782→        paymentTerm: paymentTerms || order.paymentTerms || '30',
   783→   783→        paymentMode: paymentMode || order.paymentMode || null,
   784→   784→        
   785→   785→        internalNotes: internalNotes || order.internalNotes || null,
   786→   786→        customerNotes: customerNotes || order.customerNotes || null,
   787→   787→        
   788→   788→        lines: {
   789→   789→          create: invoiceLines
   790→   790→        }
   791→   791→      },
   792→   792→      include: {
   793→   793→        partner: { select: { id: true, name: true } },
   794→   794→        lines: { include: { product: { select: { id: true, name: true } } } }
   795→   795→      }
   796→   796→    });
   797→   797→
   798→   798→    // Update sales order lines invoiced quantities
   799→   799→    for (const invLine of invoiceLines) {
   800→   800→      const soLine = linesToInvoice.find(l => l.productId === invLine.productId);
   801→   801→      if (soLine) {
   802→   802→        await tx.salesOrderLine.update({
   803→   803→          where: { id: soLine.id },
   804→   804→          data: {
   805→   805→            quantityInvoiced: soLine.quantityInvoiced + invLine.quantity
   806→   806→          }
   807→   807→        });
   808→   808→      }
   809→   809→    }
   810→   810→
   811→   811→    // Update sales order invoiced amount
   812→   812→    const newAmountInvoiced = order.amountInvoiced + amountTotal;
   813→   813→    let newStatus = order.status;
   814→   814→    
   815→   815→    // Check if fully invoiced
   816→   816→    if (Math.abs(newAmountInvoiced - order.amountTotal) < 0.01) {
   817→   817→      newStatus = order.status === 'delivered' ? 'invoiced' : 'invoiced';
   818→   818→    }
   819→   819→
   820→   820→    await tx.salesOrder.update({
   821→   821→      where: { id },
   822→   822→      data: {
   823→   823→        amountInvoiced: Math.round(newAmountInvoiced * 100) / 100,
   824→   824→        status: newStatus
   825→   825→      }
   826→   826→    });
   827→   827→
   828→   828→    return invoice;
   829→   829→  });
   830→   830→
   831→   831→  return NextResponse.json({
   832→   832→    success: true,
   833→   833→    data: result,
   834→   834→    message: `Invoice ${result.reference} created successfully from Sales Order ${order.reference}`
   835→   835→  }, { status: 201 });
   836→   836→}
