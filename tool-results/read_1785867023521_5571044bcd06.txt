   206→function formatCurrency(amount: number): string {
   207→  return new Intl.NumberFormat('fr-DZ', {
   208→    style: 'currency',
   209→    currency: 'DZD',
   210→    minimumFractionDigits: 0,
   211→    maximumFractionDigits: 0,
   212→  }).format(amount)
   213→}
   214→
   215→function formatDate(dateString: string): string {
   216→  const date = new Date(dateString)
   217→  return new Intl.DateTimeFormat('fr-FR', {
   218→    day: '2-digit',
   219→    month: '2-digit',
   220→    year: 'numeric',
   221→  }).format(date)
   222→}

   369→interface PurchaseOrderModalProps {
   370→  open: boolean
   371→  onClose: () => void
   372→  onSubmit: (data: CreatePOFormData) => Promise<void>
   373→  editingOrder?: PurchaseOrder | null
   374→  loading: boolean
   375→}
   376→
   377→interface CreatePOFormData {
   378→  partnerId: string
   379→  expectedDate?: string
   380→  paymentTerms?: string
   381→  incoterm?: string
   382→  internalNotes?: string
   383→  supplierNotes?: string
   384→  lines: PurchaseOrderLine[]
   385→}
   386→
   387→export function PurchaseOrderModal({
   388→  open,
   389→  onClose,
   390→  onSubmit,
   391→  editingOrder,
   392→  loading,
   393→}: PurchaseOrderModalProps) {
   394→  const [partners, setPartners] = useState<Partner[]>([])
   395→  const [products, setProducts] = useState<Product[]>([])
   396→  const [loadingPartners, setLoadingPartners] = useState(false)
   397→  const [loadingProducts, setLoadingProducts] = useState(false)
   398→
   399→  // Form state
   400→  const [formData, setFormData] = useState<CreatePOFormData>({
   401→    partnerId: '',
   402→    expectedDate: '',
   403→    paymentTerms: '30',
   404→    incoterm: '',
   405→    internalNotes: '',
   406→    supplierNotes: '',
   407→    lines: [{ productId: '', quantity: 1, unitPrice: 0, tvaRate: 19 }],
   408→  })
   409→
   410→  // Load partners and products on mount
   411→  useEffect(() => {
   412→    if (open) {
   413→      loadPartners()
   414→      loadProducts()
   415→      
   416→      if (editingOrder) {
   417→        setFormData({
   418→          partnerId: editingOrder.partnerId,
   419→          expectedDate: editingOrder.expectedDate
   420→            ? new Date(editingOrder.expectedDate).toISOString().split('T')[0]
   421→            : '',
   422→          paymentTerms: editingOrder.paymentTerms || '30',
   423→          incoterm: editingOrder.incoterm || '',
   424→          internalNotes: editingOrder.internalNotes || '',
   425→          supplierNotes: editingOrder.supplierNotes || '',
   426→          lines:
   427→            editingOrder.lines.length > 0
   428→              ? editingOrder.lines.map((l) => ({
   429→                  id: l.id,
   430→                  productId: l.productId,
   431→                  description: l.description,
   432→                  quantity: l.quantity,
   433→                  unitPrice: l.unitPrice,
   434→                  discountRate: l.discountRate,
   435→                  tvaRate: l.tvaRate,
   436→                }))
   437→              : [{ productId: '', quantity: 1, unitPrice: 0, tvaRate: 19 }],
   438→        })
   439→      } else {
   440→        setFormData({
   441→          partnerId: '',
   442→          expectedDate: '',
   443→          paymentTerms: '30',
   444→          incoterm: '',
   445→          internalNotes: '',
   446→          supplierNotes: '',
   447→          lines: [{ productId: '', quantity: 1, unitPrice: 0, tvaRate: 19 }],
   448→        })
   449→      }
   450→    }
   451→  }, [open, editingOrder])
   452→
   453→  async function loadPartners() {
   454→    setLoadingPartners(true)
   455→    try {
   456→      const res = await fetch('/api/partners?type=supplier')
   457→      const json = await res.json()
   458→      if (json.success) setPartners(json.data)
   459→    } catch (error) {
   460→      console.error('Error loading partners:', error)
   461→    } finally {
   462→      setLoadingPartners(false)
   463→    }
   464→  }
   465→
   466→  async function loadProducts() {
   467→    setLoadingProducts(true)
   468→    try {
   469→      const res = await fetch('/api/products?limit=100')
   470→      const json = await res.json()
   471→      if (json.success) setProducts(json.data)
   472→    } catch (error) {
   473→      console.error('Error loading products:', error)
   474→    } finally {
   475→      setLoadingProducts(false)
   476→    }
   477→  }
   478→
   479→  function updateField(field: keyof CreatePOFormData, value: string) {
   480→    setFormData((prev) => ({ ...prev, [field]: value }))
   481→  }
   482→
   483→  function updateLine(index: number, field: keyof PurchaseOrderLine, value: any) {
   484→    setFormData((prev) => ({
   485→      ...prev,
   486→      lines: prev.lines.map((line, i) =>
   487→        i === index ? { ...line, [field]: value } : line
   488→      ),
   489→    }))
   490→  }
   491→
   492→  function addLine() {
   493→    setFormData((prev) => ({
   494→      ...prev,
   495→      lines: [
   496→        ...prev.lines,
   497→        { productId: '', quantity: 1, unitPrice: 0, tvaRate: 19 },
   498→      ],
   499→    }))
   500→  }
   501→
   502→  function removeLine(index: number) {
   503→    setFormData((prev) => ({
   504→      ...prev,
   505→      lines: prev.lines.filter((_, i) => i !== index),
   506→    }))
   507→  }
   508→
   509→  // Calculate totals
   510→  const totals = useMemo(() => {
   511→    let amountUntaxed = 0
   512→    let amountTax = 0
   513→
   514→    for (const line of formData.lines) {
   515→      const qty = line.quantity || 0
   516→      const price = line.unitPrice || 0
   517→      const discount = line.discountRate || 0
   518→      const tvaRate = line.tvaRate ?? 19
   519→
   520→      const lineTotal = qty * price * (1 - discount / 100)
   521→      amountUntaxed += lineTotal
   522→      amountTax += lineTotal * (tvaRate / 100)
   523→    }
   524→
   525→    return {
   526→      amountUntaxed: Math.round(amountUntaxed * 100) / 100,
   527→      amountTax: Math.round(amountTax * 100) / 100,
   528→      amountTotal: Math.round((amountUntaxed + amountTax) * 100) / 100,
   529→    }
   530→  }, [formData.lines])
   531→
   532→  async function handleSubmit(e: React.FormEvent) {
   533→    e.preventDefault()
   534→
   535→    // Validate
   536→    if (!formData.partnerId) {
   537→      toast.error('Veuillez sélectionner un fournisseur')
   538→      return
   539→    }
   540→
   541→    const validLines = formData.lines.filter(
   542→      (l) => l.productId && l.quantity > 0 && l.unitPrice >= 0
   543→    )
   544→    if (validLines.length === 0) {
   545→      toast.error('Veuillez ajouter au moins une ligne de commande valide')
   546→      return
   547→    }
   548→
   549→    await onSubmit({
   550→      ...formData,
   551→      lines: validLines.map((l) => ({
   552→        ...l,
   553→        product: undefined,
   554→      })),
   555→    })
   556→  }
   557→
   558→  return (
   559→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   560→      <DialogContent className="max-w-4xl max-h-[90vh] overflow-y-auto">
   561→        <DialogHeader>
   562→          <DialogTitle>
   563→            {editingOrder ? `Modifier ${editingOrder.reference}` : 'Nouvelle Commande d\'Achat'}
   564→          </DialogTitle>
   565→          <DialogDescription>
   566→            {editingOrder
   567→              ? 'Modifiez les informations de la commande d\'achat'
   568→              : 'Créez une nouvelle commande d\'achat auprès d\'un fournisseur'}
   569→          </DialogDescription>
   570→        </DialogHeader>
   571→
   572→        <form onSubmit={handleSubmit} className="space-y-6">
   573→          {/* Supplier Selection */}
   574→          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   575→            <div className="space-y-2">
   576→              <Label htmlFor="partner">Fournisseur *</Label>
   577→              <Select
   578→                value={formData.partnerId}
   579→                onValueChange={(v) => updateField('partnerId', v)}
   580→                disabled={loadingPartners}
   581→              >
   582→                <SelectTrigger>
   583→                  <SelectValue placeholder="Sélectionner un fournisseur" />
   584→                </SelectTrigger>
   585→                <SelectContent>
   586→                  {partners.map((p) => (
   587→                    <SelectItem key={p.id} value={p.id}>
   588→                      {p.displayName || p.name}
   589→                    </SelectItem>
   590→                  ))}
   591→                </SelectContent>
   592→              </Select>
   593→            </div>
   594→
   595→            <div className="space-y-2">
   596→              <Label htmlFor="expectedDate">Date de livraison prévue</Label>
   597→              <Input
   598→                id="expectedDate"
   599→                type="date"
   600→                value={formData.expectedDate}
   601→                onChange={(e) => updateField('expectedDate', e.target.value)}
   602→              />
   603→            </div>
   604→
   605→            <div className="space-y-2">
   606→              <Label htmlFor="paymentTerms">Conditions de paiement</Label>
   607→              <Select
   608→                value={formData.paymentTerms}
   609→                onValueChange={(v) => updateField('paymentTerms', v)}
   610→              >
   611→                <SelectTrigger>
   612→                  <SelectValue placeholder="Sélectionner" />
   613→                </SelectTrigger>
   614→                <SelectContent>
   615→                  <SelectItem value="15">15 jours</SelectItem>
   616→                  <SelectItem value="30">30 jours</SelectItem>
   617→                  <SelectItem value="45">45 jours</SelectItem>
   618→                  <SelectItem value="60">60 jours</SelectItem>
   619→                  <SelectItem value="90">90 jours</SelectItem>
   620→                </SelectContent>
   621→              </Select>
   622→            </div>
   623→
   624→            <div className="space-y-2">
   625→              <Label htmlFor="incoterm">Incoterm</Label>
   626→              <Select
   627→                value={formData.incoterm}
   628→                onValueChange={(v) => updateField('incoterm', v)}
   629→              >
   630→                <SelectTrigger>
   631→                  <SelectValue placeholder="Sélectionner" />
   632→                </SelectTrigger>
   633→                <SelectContent>
   634→                  <SelectItem value="">Aucun</SelectItem>
   635→                  <SelectItem value="EXW">EXW (Départ usine)</SelectItem>
   636→                  <SelectItem value="FCA">FCA</SelectItem>
   637→                  <SelectItem value="FOB">FOB</SelectItem>
   638→                  <SelectItem value="CIF">CIF</SelectItem>
   639→                  <SelectItem value="DDP">DDP (Rendu destination)</SelectItem>
   640→                </SelectContent>
   641→              </Select>
   642→            </div>
   643→          </div>
   644→
   645→          {/* Order Lines */}
   646→          <div className="space-y-4">

  1235→  // Status change handler
  1236→  async function handleStatusChange(id: string, action: string) {
  1237→    setActionLoading(true)
  1238→    try {
  1239→      let url = `/api/purchases/${id}`
  1240→      let options: RequestInit = {}
  1241→
  1242→      switch (action) {
  1243→        case 'send':
  1244→          url += '?action=send'
  1245→          options.method = 'POST'
  1246→          break
  1247→        case 'confirm':
  1248→          url += '?action=confirm'
  1249→          options.method = 'POST'
  1250→          break
  1251→        case 'receive':
  1252→          // For receive, we need to send the lines data
  1253→          options.method = 'POST'
  1254→          options.headers = { 'Content-Type': 'application/json' }
  1255→          options.body = JSON.stringify({ 
  1256→            action: 'receive',
  1257→            lines: (order?.lines || []).map((l: PurchaseOrderLine) => ({
  1258→              lineId: l.id,
  1259→              quantity: l.quantity - (l as any).quantityReceived
  1260→            })).filter((l: any) => l.quantity > 0)
  1261→          })
  1262→          break
  1263→        case 'bill':
  1264→          url += '?action=bill'
  1265→          options.method = 'POST'
  1266→          options.headers = { 'Content-Type': 'application/json' }
  1267→          options.body = JSON.stringify({})
  1268→          break
  1269→        case 'cancel':
  1270→          options.method = 'DELETE'
  1271→          break
  1272→        default:
  1273→          throw new Error(`Action inconnue: ${action}`)
  1274→      }
  1275→
  1276→      const response = await fetch(url, options)
  1277→      const result = await response.json()
  1278→
  1279→      if (result.success) {
  1280→        // Show SCF journal entry info if bill created
  1281→        if (action === 'bill' && result.workflowInfo?.journalEntryGenerated) {
  1282→          toast.success(result.message || 'Facture fournisseur créée ✓ Écriture comptable SCF générée', { duration: 5000 })
  1283→        } else {
  1284→          toast.success(result.message || 'Statut mis à jour')
  1285→        }
  1286→        setDetailModalOpen(false)
  1287→        setSelectedOrder(null)
  1288→        fetchOrders()
  1289→      } else {
  1290→        toast.error(result.error || 'Erreur lors de la mise à jour')
  1291→      }
  1292→    } catch (err) {
  1293→      toast.error('Erreur réseau')
  1294→    } finally {
  1295→      setActionLoading(false)
  1296→    }
  1297→  }
