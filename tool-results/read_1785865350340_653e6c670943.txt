   512→export default function HrPage() {
   513→  // State
   514→  const [employees, setEmployees] = useState<Employee[]>([])
   515→  const [payrollData, setPayrollData] = useState<PayrollRecord[]>([])
   516→  const [loading, setLoading] = useState(true)
   517→  const [error, setError] = useState<string | null>(null)
   518→  
   519→  // Filters & Search
   520→  const [searchQuery, setSearchQuery] = useState('')
   521→  const [departmentFilter, setDepartmentFilter] = useState('all')
   522→  const [statusFilter, setStatusFilter] = useState('all')
   523→  
   524→  // Modal
   525→  const [selectedEmployee, setSelectedEmployee] = useState<Employee | null>(null)
   526→  const [isModalOpen, setIsModalOpen] = useState(false)
   527→
   528→  // Fetch employees
   529→  const fetchEmployees = useCallback(async () => {
   530→    try {
   531→      setLoading(true)
   532→      setError(null)
   533→      
   534→      const params = new URLSearchParams()
   535→      if (departmentFilter !== 'all') params.append('department', departmentFilter)
   536→      if (statusFilter !== 'all') params.append('status', statusFilter)
   537→      if (searchQuery) params.append('search', searchQuery)
   538→      
   539→      const response = await fetch(`/api/employees?${params.toString()}`)
   540→      const result = await response.json()
   541→      
   542→      if (!result.success) {
   543→        throw new Error(result.error || 'Failed to fetch employees')
   544→      }
   545→      
   546→      setEmployees(result.data)
   547→    } catch (err) {
   548→      setError(err instanceof Error ? err.message : 'An error occurred')
   549→    } finally {
   550→      setLoading(false)
   551→    }
   552→  }, [departmentFilter, statusFilter, searchQuery])
   553→
   554→  // Fetch payroll data
   555→  const fetchPayrollData = useCallback(async () => {
   556→    try {
   557→      const currentPeriod = new Date().toISOString().slice(0, 7) // YYYY-MM
   558→      const response = await fetch(`/api/payroll?period=${currentPeriod}`)
   559→      const result = await response.json()
   560→      
   561→      if (result.success) {
   562→        setPayrollData(result.data)
   563→      }
   564→    } catch (err) {
   565→      console.error('Failed to fetch payroll:', err)
   566→    }
   567→  }, [])
   568→
   569→  // Initial load
   570→  useEffect(() => {
   571→    fetchEmployees()
   572→    fetchPayrollData()
   573→  }, [])
   574→
   575→  // Refetch when filters change (debounced)
   576→  useEffect(() => {
   577→    const timer = setTimeout(() => {
   578→      if (!loading) {
   579→        fetchEmployees()
   580→      }
   581→    }, 300)
   582→    
   583→    return () => clearTimeout(timer)
   584→  }, [departmentFilter, statusFilter, searchQuery, loading, fetchEmployees])
   585→
   586→  // Computed values
   587→  const stats: HrStats = useMemo(() => {
   588→    const total = employees.length
   589→    const active = employees.filter(e => e.employeeStatus === 'active').length
   590→    const onLeave = employees.filter(e => e.employeeStatus === 'on_leave').length
   591→    
   592→    // Calculate payroll mass from fetched data or use sum of base salaries
