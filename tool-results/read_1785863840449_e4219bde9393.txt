   298→function formatDZD(amount: number): string {
   299→  return new Intl.NumberFormat('fr-DZ', {
   300→    minimumFractionDigits: 0,
   301→    maximumFractionDigits: 2,
   302→  }).format(amount)
   303→}
   304→
   305→function formatDate(dateString: string): string {
   306→  return new Date(dateString).toLocaleDateString('fr-DZ', {
   307→    day: '2-digit',
   308→    month: '2-digit',
   309→    year: 'numeric',
   310→  })
   311→}
   312→
   313→function formatDateTime(dateString: string): string {
   314→  return new Date(dateString).toLocaleDateString('fr-DZ', {
   315→    day: '2-digit',
   316→    month: '2-digit',
   317→    year: 'numeric',
   318→    hour: '2-digit',
   319→    minute: '2-digit',
   320→  })
   321→}
   322→
   323→function getDaysUntil(dateString: string): number {
   324→  const target = new Date(dateString)
   325→  const now = new Date()
   326→  const diffTime = target.getTime() - now.getTime()
   327→  return Math.ceil(diffTime / (1000 * 60 * 60 * 24))
   328→}
   329→
   330→function isExpired(dateString: string): boolean {
   331→  return getDaysUntil(dateString) < 0
   332→}

   337→function SalesOrderStatusBadge({ status }: { status: string }) {
   338→  const config = SALES_ORDER_STATUSES[status as keyof typeof SALES_ORDER_STATUSES] || SALES_ORDER_STATUSES.draft
   339→  const Icon = config.icon
   340→  
   341→  return (
   342→    <Badge variant="outline" className={`${config.color} gap-1.5`}>
   343→      <Icon className="w-3 h-3" />
   344→      {config.label}
   345→    </Badge>
   346→  )
   347→}
   348→
   349→function QuotationStatusBadge({ status, validUntil }: { status: string; validUntil?: string }) {
   350→  let config = QUOTATION_STATUSES[status as keyof typeof QUOTATION_STATUSES]
   351→  
   352→  // Auto-detect expired status
   353→  if (status === 'sent' && validUntil && isExpired(validUntil)) {
   354→    config = QUOTATION_STATUSES.expired
   355→  }
   356→  
   357→  if (!config) config = QUOTATION_STATUSES.draft
   358→  const Icon = config.icon
   359→  
   360→  return (
   361→    <Badge variant="outline" className={`${config.color} gap-1.5`}>
   362→      <Icon className="w-3 h-3" />
   363→      {config.label}
   364→    </Badge>
   365→  )
   366→}
