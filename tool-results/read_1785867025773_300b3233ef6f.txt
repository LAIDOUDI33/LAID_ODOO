   307→   307→    day: '2-digit',
   308→   308→    month: '2-digit',
   309→   309→    year: 'numeric',

   313→   313→function formatDateTime(dateString: string): string {
   314→   314→  return new Date(dateString).toLocaleDateString('fr-DZ', {
   315→   315→    day: '2-digit',
   316→   316→    month: '2-digit',
   317→   317→    year: 'numeric',
   318→   318→    hour: '2-digit',
   319→   319→    minute: '2-digit',
   320→   320→  })
   321→   321→}
   322→   322→
   323→   323→function getDaysUntil(dateString: string): number {
   324→   324→  const target = new Date(dateString)
   325→   325→  const now = new Date()
   326→   326→  const diffTime = target.getTime() - now.getTime()
   327→   327→  return Math.ceil(diffTime / (1000 * 60 * 60 * 24))
   328→   328→}
   329→   329→
   330→   330→function isExpired(dateString: string): boolean {
   331→   331→  return getDaysUntil(dateString) < 0
   332→   332→}
