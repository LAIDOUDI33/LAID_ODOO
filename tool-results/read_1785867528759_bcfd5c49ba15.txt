   149→// Format currency in DZD
   150→function formatDZD(amount: number): string {
   151→  return new Intl.NumberFormat('fr-DZ', {
   152→    style: 'currency',
   153→    currency: 'DZD',
   154→    minimumFractionDigits: 0,
   155→    maximumFractionDigits: 2,
   156→  }).format(amount)
   157→}
   158→
   159→// Format number
   160→function formatNumber(num: number): string {
   161→  return new Intl.NumberFormat('fr-DZ').format(num)
   162→}

   164→// Get stock status badge
   165→function getStockStatus(quantity: number, minQty: number) {
   166→  if (quantity === 0) {
   167→    return <Badge variant="outline" className="bg-gray-100 text-gray-700 border-gray-200">Rupture</Badge>
   168→  }
   169→  if (quantity <= minQty * 0.5 && minQty > 0) {
   170→    return <Badge variant="destructive">Critique</Badge>
   171→  }
   172→  if (quantity <= minQty && minQty > 0) {
   173→    return <Badge variant="outline" className="bg-yellow-100 text-yellow-700 border-yellow-200 dark:bg-yellow-900/30 dark:text-yellow-400">Bas</Badge>
   174→  }
   175→  return <Badge variant="outline" className="bg-green-100 text-green-700 border-green-200 dark:bg-green-900/30 dark:text-green-400">En stock</Badge>
   176→}

   557→// Stock Adjustment Modal
   558→function StockAdjustModal({
   559→  open,
   560→  onClose,
   561→  onAdjust,
   562→  stockItem,
   563→  warehouses,
   564→  loading
   565→}: {
   566→  open: boolean
   567→  onClose: () => void
   568→  onAdjust: (data: any) => void
   569→  stockItem?: StockLevel | null
   570→  warehouses: Warehouse[]
   571→  loading: boolean
   572→}) {
   573→  const [form, setForm] = useState({
   574→    warehouseId: '',
   575→    quantity: '0',
   576→    type: 'in' as 'in' | 'out',
   577→    notes: ''
   578→  })
   579→
   580→  useEffect(() => {
   581→    if (stockItem) {
   582→      setForm({
   583→        warehouseId: stockItem.warehouseId,
   584→        quantity: '0',
   585→        type: 'in',
   586→        notes: ''
   587→      })
   588→    } else if (warehouses.length > 0) {
   589→      setForm({
   590→        warehouseId: warehouses[0]?.id || '',
   591→        quantity: '0',
   592→        type: 'in',
   593→        notes: ''
   594→      })
   595→    }
   596→  }, [stockItem, open, warehouses])
   597→
   598→  const handleSubmit = (e: React.FormEvent) => {
   599→    e.preventDefault()
   600→    onAdjust({
   601→      productId: stockItem?.productId || '',
   602→      warehouseId: form.warehouseId,
   603→      locationId: stockItem?.locationId || null,
