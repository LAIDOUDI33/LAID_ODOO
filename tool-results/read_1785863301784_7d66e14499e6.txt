   123→// ============================================================
   124→// Status Transition Validation
   125→// ============================================================
   126→
   127→const VALID_STATUS_TRANSITIONS: Record<string, string[]> = {
   128→  draft: ['sent', 'confirmed', 'cancelled'],
   129→  sent: ['confirmed', 'draft', 'cancelled'],
   130→  confirmed: ['received', 'cancelled'],
   131→  received: ['billed', 'done'],
   132→  billed: ['done', 'done'],
   133→  done: [],
   134→  cancelled: ['draft'], // Can be reinstated to draft
   135→};
   136→
   137→function canTransition(currentStatus: string, newStatus: string): boolean {
   138→  const allowedTransitions = VALID_STATUS_TRANSITIONS[currentStatus] || [];
   139→  return allowedTransitions.includes(newStatus);
   140→}

   143→// PUT /api/purchases/[id] - Update Purchase Order
   144→// ============================================================
   145→
   146→export async function PUT(
   147→  request: NextRequest,
   148→  { params }: { params: Promise<{ id: string }> }
   149→) {
   150→  try {
   151→    const { id } = await params;
   152→    const body: UpdatePurchaseOrderInput = await request.json();
   153→    
   154→    // Check if PO exists
   155→    const existingPO = await db.purchaseOrder.findUnique({
   156→      where: { id },
   157→      include: {
   158→        lines: true,
   159→      },
   160→    });
   161→    
   162→    if (!existingPO) {
   163→      return NextResponse.json(
   164→        {
   165→          success: false,
   166→          error: 'Commande d\'achat non trouvée',
   167→        },
   168→        { status: 404 }
   169→      );
   170→    }
   171→    
   172→    // Check if PO can be updated (only draft or cancelled status)
   173→    if (!['draft', 'cancelled'].includes(existingPO.status)) {
   174→      return NextResponse.json(
   175→        {
   176→          success: false,
   177→          error: `Impossible de modifier une commande avec le statut '${existingPO.status}'. Seules les commandes en brouillon ou annulées peuvent être modifiées.`,
   178→        },
   179→        { status: 400 }
   180→      );
   181→    }
   182→    
   183→    // Validate partner if provided
   184→    if (body.partnerId && body.partnerId !== existingPO.partnerId) {
   185→      const partner = await db.partner.findUnique({
   186→        where: { id: body.partnerId },
   187→      });
   188→      
   189→      if (!partner) {
   190→        return NextResponse.json(
   191→          {
   192→            success: false,
   193→            error: 'Fournisseur non trouvé',
   194→          },
   195→          { status: 404 }
   196→        );
   197→      }
   198→      
   199→      if (partner.type === 'customer') {
   200→        return NextResponse.json(
   201→          {
   202→            success: false,
   203→            error: 'Ce partenaire n\'est pas un fournisseur',
   204→          },
   205→          { status: 400 }
   206→        );
   207→      }
   208→    }
   209→    
   210→    // Validate warehouse if provided
   211→    if (body.warehouseId && body.warehouseId !== existingPO.warehouseId) {
   212→      const warehouse = await db.warehouse.findUnique({
   213→        where: { id: body.warehouseId },
   214→      });
   215→      
   216→      if (!warehouse) {
   217→        return NextResponse.json(
   218→          {
   219→            success: false,
   220→            error: 'Entrepôt non trouvé',
   221→          },
   222→          { status: 404 }
   223→        );
   224→      }
   225→    }
   226→    
   227→    // Prepare update data
   228→    const updateData: Record<string, unknown> = {
   229→      updatedAt: new Date(),
   230→    };
   231→    
   232→    if (body.partnerId) updateData.partnerId = body.partnerId;
   233→    if (body.date) updateData.date = new Date(body.date);
   234→    if (body.expectedDate !== undefined) {
   235→      updateData.expectedDate = body.expectedDate ? new Date(body.expectedDate) : null;
   236→    }
   237→    if (body.paymentTerms !== undefined) updateData.paymentTerms = body.paymentTerms;
   238→    if (body.paymentMode !== undefined) updateData.paymentMode = body.paymentMode;
   239→    if (body.incoterm !== undefined) updateData.incoterm = body.incoterm;
   240→    if (body.shippingAddress !== undefined) updateData.shippingAddress = body.shippingAddress;
   241→    if (body.warehouseId !== undefined) updateData.warehouseId = body.warehouseId;
   242→    if (body.internalNotes !== undefined) updateData.internalNotes = body.internalNotes;
   243→    if (body.supplierNotes !== undefined) updateData.supplierNotes = body.supplierNotes;
   244→    
   245→    // If lines are provided, recalculate everything
   246→    if (body.lines && body.lines.length > 0) {
   247→      const processedLines = [];
   248→      
   249→      for (let i = 0; i < body.lines.length; i++) {
   250→        const line = body.lines[i];
   251→        
   252→        // Validate required fields
   253→        if (!line.productId) {
   254→          return NextResponse.json(
   255→            {
   256→              success: false,
   257→              error: `Le produit est requis pour la ligne ${i + 1}`,
   258→            },
   259→            { status: 400 }
   260→          );
   261→        }
   262→        
   263→        if (!line.quantity || line.quantity <= 0) {
   264→          return NextResponse.json(
   265→            {
   266→              success: false,
   267→              error: `La quantité doit être supérieure à 0 pour la ligne ${i + 1}`,
   268→            },
   269→            { status: 400 }
   270→          );
   271→        }
   272→        
   273→        if (line.unitPrice === undefined || line.unitPrice < 0) {
   274→          return NextResponse.json(
   275→            {
   276→              success: false,
   277→              error: `Le prix unitaire doit être positif pour la ligne ${i + 1}`,
   278→            },
   279→            { status: 400 }
   280→          );
   281→        }
   282→        
   283→        // Validate product exists
   284→        const product = await db.product.findUnique({
   285→          where: { id: line.productId },
   286→        });
   287→        
   288→        if (!product) {
   289→          return NextResponse.json(
   290→            {
   291→              success: false,
   292→              error: `Produit non trouvé pour la ligne ${i + 1}`,
   293→            },
   294→            { status: 404 }
   295→          );
   296→        }
   297→        
   298→        // Validate TVA rate
   299→        const tvaRate = line.tvaRate ?? product.tvaRate ?? 19;
   300→        if (!isValidTVARate(tvaRate)) {
   301→          return NextResponse.json(
   302→            {
   303→              success: false,
   304→              error: `Taux TVA invalide pour la ligne ${i + 1}. Taux autorisés: 0%, 9%, 19%`,
   305→            },
   306→            { status: 400 }
   307→          );
   308→        }
   309→        
   310→        // Calculate amounts
   311→        const amounts = calculateLineAmounts({
   312→          ...line,
   313→          tvaRate,
   314→        });
   315→        
   316→        processedLines.push({
   317→          id: line.id, // Keep existing line ID if updating
   318→          productId: line.productId,
   319→          description: line.description || product.name,
   320→          quantity: line.quantity,
   321→          unitPrice: line.unitPrice,
   322→          discountRate: line.discountRate || 0,
   323→          tvaRate,
   324→          ...amounts,
   325→        });
   326→      }
   327→      
   328→      // Calculate order totals
   329→      const totals = calculateOrderTotals(processedLines);
   330→      updateData.amountUntaxed = totals.amountUntaxed;
   331→      updateData.amountTax = totals.amountTax;
   332→      updateData.amountTotal = totals.amountTotal;
   333→      
   334→      // Perform update in transaction
   335→      const updatedOrder = await db.$transaction(async (tx) => {
   336→        // Delete existing lines
   337→        await tx.purchaseOrderLine.deleteMany({
   338→          where: { purchaseOrderId: id },
   339→        });
   340→        
   341→        // Create new lines
   342→        const lines = await Promise.all(
   343→          processedLines.map((line) =>
   344→            tx.purchaseOrderLine.create({
   345→              data: {
   346→                purchaseOrderId: id,
   347→                productId: line.productId,
   348→                description: line.description,
   349→                quantity: line.quantity,
   350→                unitPrice: line.unitPrice,
   351→                discountRate: line.discountRate,
   352→                tvaRate: line.tvaRate,
   353→                amountUntaxed: line.amountUntaxed,
   354→                amountTax: line.amountTax,
   355→                amountTotal: line.amountTotal,
   356→                quantityReceived: 0,
   357→                quantityInvoiced: 0,
   358→              },
   359→            })
   360→          )
   361→        );
   362→        
   363→        // Update order header
   364→        const order = await tx.purchaseOrder.update({
   365→          where: { id },
   366→          data: updateData,
   367→        });
   368→        
   369→        return { ...order, lines };
   370→      });
   371→      
   372→      // Return full updated order
   373→      const fullOrder = await getPurchaseOrderWithDetails(id);
   374→      
   375→      return NextResponse.json({
   376→        success: true,
   377→        data: fullOrder,
   378→        message: `Commande d'achat ${existingPO.reference} mise à jour avec succès`,
   379→      });
   380→    } else {
   381→      // Just update header without changing lines
   382→      const updatedOrder = await db.purchaseOrder.update({
   383→        where: { id },
   384→        data: updateData,
   385→      });
   386→      
   387→      const fullOrder = await getPurchaseOrderWithDetails(id);
   388→      
   389→      return NextResponse.json({
   390→        success: true,
   391→        data: fullOrder,
   392→        message: `Commande d'achat ${existingPO.reference} mise à jour avec succès`,
   393→      });
   394→    }
   395→  } catch (error) {
   396→    console.error('Error updating purchase order:', error);
   397→    return NextResponse.json(
   398→      {
   399→        success: false,
   400→        error: 'Erreur lors de la mise à jour de la commande d\'achat',
   401→      },
   402→      { status: 500 }
   403→    );
   404→  }
   405→}

   408→// DELETE /api/purchases/[id] - Cancel/Delete Purchase Order
   409→// ============================================================
   410→
   411→export async function DELETE(
   412→  request: NextRequest,
   413→  { params }: { params: Promise<{ id: string }> }
   414→) {
   415→  try {
   416→    const { id } = await params;
   417→    
   418→    // Check if PO exists
   419→    const existingPO = await db.purchaseOrder.findUnique({
   420→      where: { id },
   421→      include: {
   422→        receiptItems: true,
   423→        bills: true,
   424→      },
   425→    });
   426→    
   427→    if (!existingPO) {
   428→      return NextResponse.json(
   429→        {
   430→          success: false,
   431→          error: 'Commande d\'achat non trouvée',
   432→        },
   433→        { status: 404 }
   434→      );
   435→    }
   436→    
   437→    // Check if already cancelled
   438→    if (existingPO.status === 'cancelled') {
   439→      return NextResponse.json(
   440→        {
   441→          success: false,
   442→          error: 'Cette commande est déjà annulée',
   443→        },
   444→        { status: 400 }
   445→      );
   446→    }
   447→    
   448→    // Check if goods have been received
   449→    if (existingPO.receiptItems.length > 0) {
   450→      return NextResponse.json(
   451→        {
   452→          success: false,
   453→          error: 'Impossible d\'annuler cette commande car des marchandises ont déjà été reçues. Veuillez d\'abord effectuer un retour de stock.',
   454→        },
   455→        { status: 400 }
   456→      );
   457→    }
   458→    
   459→    // Check if bills exist
   460→    if (existingPO.bills.length > 0) {
   461→      return NextResponse.json(
   462→        {
   463→          success: false,
   464→          error: 'Impossible d\'annuler cette commande car des factures fournisseurs y sont liées.',
   465→        },
   466→        { status: 400 }
   467→      );
   468→    }
   469→    
   470→    // Cancel the order (soft delete by setting status to cancelled)
   471→    const cancelledOrder = await db.purchaseOrder.update({
   472→      where: { id },
   473→      data: {
   474→        status: 'cancelled',
   475→      },
   476→    });
   477→    
   478→    return NextResponse.json({
   479→      success: true,
   480→      data: cancelledOrder,
   481→      message: `Commande d'achat ${existingPO.reference} annulée avec succès`,
   482→    });
   483→  } catch (error) {
   484→    console.error('Error cancelling purchase order:', error);
   485→    return NextResponse.json(
   486→      {
   487→        success: false,
   488→        error: 'Erreur lors de l\'annulation de la commande d\'achat',
   489→      },
   490→      { status: 500 }
   491→    );
   492→  }
   493→}

   529→        error: 'Erreur lors du traitement de la requête',

   540→async function handleReceiveGoods(request: NextRequest, poId: string): Promise<NextResponse> {
   541→  try {
   542→    const body: ReceiveGoodsInput = await request.json();
   543→    
   544→    // Validate input
   545→    if (!body.lines || body.lines.length === 0) {
   546→      return NextResponse.json(
   547→        {
   548→          success: false,
   549→          error: 'Au moins une ligne de réception est requise',
   550→        },
   551→        { status: 400 }
   552→      );
   553→    }
   554→    
   555→    // Get PO with full details
   556→    const po = await db.purchaseOrder.findUnique({
   557→      where: { id: poId },
   558→      include: {
   559→        lines: {
   560→          include: {
   561→            product: true,
   562→          },
   563→        },
   564→        warehouse: true,
   565→        company: true,
   566→      },
   567→    });
   568→    
   569→    if (!po) {
   570→      return NextResponse.json(
   571→        {
   572→          success: false,
   573→          error: 'Commande d\'achat non trouvée',
   574→        },
   575→        { status: 404 }
   576→      );
   577→    }
   578→    
   579→    // Check if PO can receive goods
   580→    if (!['confirmed', 'sent', 'draft'].includes(po.status) && po.status !== 'received') {
   581→      return NextResponse.json(
   582→        {
   583→          success: false,
   584→          error: `La commande doit être confirmée avant de pouvoir recevoir des marchandises. Statut actuel: ${po.status}`,
   585→        },
   586→        { status: 400 }
   587→      );
   588→    }
   589→    
   590→    // Check if warehouse is set
   591→    if (!po.warehouseId) {
   592→      return NextResponse.json(
   593→        {
   594→          success: false,
   595→          error: 'Aucun entrepôt défini pour cette commande. Veuillez définir l\'entrepôt de réception.',
   596→        },
   597→        { status: 400 }
   598→      );
   599→    }
   600→    
   601→    // Process each received line
   602→    const movements = [];
   603→    const updatedLines = [];
   604→    let totalReceived = 0;
   605→    
   606→    for (const recvLine of body.lines) {
   607→      // Find the corresponding PO line
   608→      const poLine = po.lines.find((l) => l.id === recvLine.lineId);
   609→      
   610→      if (!poLine) {
   611→        return NextResponse.json(
   612→          {
   613→            success: false,
   614→            error: `Ligne de commande ${recvLine.lineId} non trouvée`,
   615→          },
   616→          { status: 404 }
   617→        );
   618→      }
   619→      
   620→      // Validate quantity
   621→      if (!recvLine.quantity || recvLine.quantity <= 0) {
   622→        return NextResponse.json(
   623→          {
   624→            success: false,
   625→            error: `La quantité reçue doit être supérieure à 0 pour le produit ${poLine.product.name}`,
   626→          },
   627→          { status: 400 }
   628→        );
   629→      }
   630→      
   631→      // Check not over-receiving
   632→      const maxReceivable = poLine.quantity - poLine.quantityReceived;
   633→      if (recvLine.quantity > maxReceivable) {
   634→        return NextResponse.json(
   635→          {
   636→            success: false,
   637→            error: `Quantité reçue (${recvLine.quantity}) supérieure au restant à recevoir (${maxReceivable}) pour ${poLine.product.name}`,
   638→          },
   639→          { status: 400 }
   640→        );
   641→      }
   642→      
   643→      // Calculate cost for this reception
   644→      const lineUnitCost = poLine.unitPrice * (1 - poLine.discountRate / 100);
   645→      const totalCost = Math.round(recvLine.quantity * lineUnitCost * 100) / 100;
   646→      
   647→      movements.push({
   648→        productId: poLine.productId,
   649→        quantity: recvLine.quantity,
   650→        unitCost: lineUnitCost,
   651→        totalCost,
   652→        locationId: recvLine.locationId || null,
   653→        lineId: poLine.id,
   654→      });
   655→      
   656→      updatedLines.push({
   657→        lineId: poLine.id,
   658→        quantityReceived: poLine.quantityReceived + recvLine.quantity,
   659→        amountReceived: Math.round((poLine.quantityReceived + recvLine.quantity) * lineUnitCost * 100) / 100,
   660→      });
   661→      
   662→      totalReceived += totalCost;
   663→    }
   664→    
   665→    // Generate movement reference
   666→    const now = new Date();
   667→    const year = now.getFullYear();
   668→    const month = String(now.getMonth() + 1).padStart(2, '0');
   669→    const movementCount = await db.stockMovement.count({
   670→      where: {
   671→        type: 'in_receipt',
   672→        date: {
   673→          gte: new Date(year, now.getMonth(), 1),
   674→          lt: new Date(year, now.getMonth() + 1, 1),
   675→        },
   676→      },
   677→    });
   678→    const movementRef = `ENT-${year}-${month}-${String(movementCount + 1).padStart(3, '0')}`;
   679→    
   680→    // Process in transaction
   681→    const result = await db.$transaction(async (tx) => {
   682→      // Create stock movements
   683→      const createdMovements = await Promise.all(
   684→        movements.map((mov) =>
   685→          tx.stockMovement.create({
   686→            data: {
   687→              reference: `${movementRef}-${mov.productId.slice(-6)}`,
   688→              date: now,
   689→              type: 'in_receipt',
   690→              quantity: mov.quantity,
   691→              unitCost: mov.unitCost,
   692→              totalCost: mov.totalCost,
   693→              notes: body.notes || `Réception commande ${po.reference}`,
   694→              productId: mov.productId,
   695→              warehouseId: po.warehouseId!,
   696→              locationId: mov.locationId,
   697→              sourceDoc: 'purchase_order',
   698→              sourceId: poId,
   699→              purchaseOrderId: poId,
   700→            },
   701→          })
   702→        )
   703→      );
   704→      
   705→      // Update PO lines quantities received
   706→      for (const ul of updatedLines) {
   707→        await tx.purchaseOrderLine.update({
   708→          where: { id: ul.lineId },
   709→          data: {
   710→            quantityReceived: ul.quantityReceived,
   711→          },
   712→        });
   713→      }
   714→      
   715→      // Update stock levels for each product
   716→      for (const mov of movements) {
   717→        // Find or create stock level
   718→        const existingStockLevel = await tx.stockLevel.findUnique({
   719→          where: {
   720→            productId_warehouseId_locationId: {
   721→              productId: mov.productId,
   722→              warehouseId: po.warehouseId!,
   723→              locationId: mov.locationId ?? '',
   724→            },
   725→          },
   726→        });
   727→        
   728→        if (existingStockLevel) {
   729→          // Update existing stock level
   730→          const newQuantity = existingStockLevel.quantity + mov.quantity;
   731→          await tx.stockLevel.update({
   732→            where: { id: existingStockLevel.id },
   733→            data: {
   734→              quantity: Math.round(newQuantity * 100) / 100,
   735→              availableQty: Math.round((newQuantity - existingStockLevel.reservedQty) * 100) / 100,
   736→            },
   737→          });
   738→        } else {
   739→          // Create new stock level
   740→          await tx.stockLevel.create({
   741→            data: {
   742→              productId: mov.productId,
   743→              warehouseId: po.warehouseId!,
   744→              locationId: mov.locationId,
   745→              quantity: mov.quantity,
   746→              reservedQty: 0,
   747→              availableQty: mov.quantity,
   748→              minQty: 0,
   749→              maxQty: 0,
   750→            },
   751→          });
   752→        }
   753→      }
   754→      
   755→      // Determine new PO status
   756→      let newStatus = po.status;
   757→      const allFullyReceived = po.lines.every(
   758→        (l) => {
   759→          const updatedL = updatedLines.find((ul) => ul.lineId === l.id);
   760→          return updatedL ? updatedL.quantityReceived >= l.quantity : l.quantityReceived >= l.quantity;
   761→        }
   762→      );
   763→      
   764→      if (allFullyReceived) {
   765→        newStatus = 'received';
   766→      } else if (updatedLines.length > 0 && po.status === 'confirmed') {
   767→        newStatus = 'received'; // Partially received
   768→      }
   769→      
   770→      // Update PO
   771→      const updatedPO = await tx.purchaseOrder.update({
   772→        where: { id: poId },
   773→        data: {
   774→          amountReceived: po.amountReceived + totalReceived,
   775→          receiptDate: now,
   776→          status: newStatus,
   777→        },
   778→      });
   779→      
   780→      return {
   781→        movements: createdMovements,
   782→        purchaseOrder: updatedPO,
   783→      };
   784→    });
   785→    
   786→    // Return full updated order
   787→    const fullOrder = await getPurchaseOrderWithDetails(poId);
   788→    
   789→    return NextResponse.json({
   790→      success: true,
   791→      data: {
   792→        purchaseOrder: fullOrder,
   793→        movements: result.movements,
   794→      },
   795→      message: `${movements.length} produit(s) reçu(s) avec succès`,
   796→    });
   797→  } catch (error) {
   798→    console.error('Error receiving goods:', error);
   799→    return NextResponse.json(
   800→      {
   801→        success: false,
   802→        error: 'Erreur lors de la réception des marchandises',
   803→      },
   804→      { status: 500 }
   805→    );
   806→  }
   807→}

   811→// ============================================================

   813→async function handleConfirmOrder(request: NextRequest, poId: string): Promise<NextResponse> {
   814→  try {
   815→    // Get PO
   816→    const po = await db.purchaseOrder.findUnique({
   817→      where: { id: poId },
   818→    });
   819→    
   820→    if (!po) {
   821→      return NextResponse.json(
   822→        {
   823→          success: false,
   824→          error: 'Commande d\'achat non trouvée',
   825→        },
   826→        { status: 404 }
   827→      );
   828→    }
   829→    
   830→    // Check current status and validate transition
   831→    const targetStatus = 'confirmed';
   832→    
   833→    if (!canTransition(po.status, targetStatus)) {
   834→      return NextResponse.json(
   835→        {
   836→          success: false,
   837→          error: `Transition de statut invalide: '${po.status}' -> '${targetStatus}'`,
   838→        },
   839→        { status: 400 }
   840→      );
   841→    }
   842→    
   843→    // Determine intermediate status based on current state
   844→    let newStatus: string;
   845→    
   846→    switch (po.status) {
   847→      case 'draft':
   848→        newStatus = 'sent'; // First send to supplier
   849→        break;
   850→      case 'sent':
   851→      case 'cancelled':
   852→        newStatus = 'confirmed'; // Then confirm
   853→        break;
   854→      default:
   855→        newStatus = targetStatus;
   856→    }
   857→    
   858→    // Update status
   859→    const updatedPO = await db.purchaseOrder.update({
   860→      where: { id: poId },
   861→      data: {
   862→        status: newStatus,
   863→      },
   864→    });
   865→    
   866→    // Return full order
   867→    const fullOrder = await getPurchaseOrderWithDetails(poId);
   868→    
   869→    const statusMessages: Record<string, string> = {
   870→      sent: 'Commande envoyée au fournisseur',
   871→      confirmed: 'Commande confirmée par le fournisseur',
   872→    };
   873→    
   874→    return NextResponse.json({
   875→      success: true,
   876→      data: fullOrder,
   877→      message: statusMessages[newStatus] || `Statut mis à jour: ${newStatus}`,
   878→    });
   879→  } catch (error) {
   880→    console.error('Error confirming purchase order:', error);
   881→    return NextResponse.json(
   882→      {
   883→        success: false,
   884→        error: 'Erreur lors de la confirmation de la commande d\'achat',
   885→      },
   886→      { status: 500 }
   887→    );
   888→  }
   889→}
