   468→
   469→function useApiData<T>(url: string, enabled = true) {
   470→  const [data, setData] = useState<T[]>([])
   471→  const [loading, setLoading] = useState(true)
   472→  const [error, setError] = useState<string | null>(null)
   473→  const [pagination, setPagination] = useState<{ page: number; limit: number; total: number; pages: number } | null>(null)
   474→  const [stats, setStats] = useState<any>(null)
   475→
   476→  const fetchData = useCallback(async () => {
   477→    if (!enabled) {
   478→      setLoading(false)
   479→      return
   480→    }
   481→    
   482→    setLoading(true)
   483→    setError(null)
   484→    
   485→    try {
   486→      const response = await fetch(url)
   487→      const result: ApiResponse<T> = await response.json()
   488→      
   489→      if (result.success) {
   490→        setData(result.data)
   491→        setPagination(result.pagination || null)
   492→        setStats(result.stats || null)
   493→      } else {
   494→        setError(result.error || 'Erreur lors du chargement des données')
   495→      }
   496→    } catch (err) {
   497→      setError('Erreur de connexion au serveur')
   498→    } finally {
   499→      setLoading(false)
   500→    }
   501→  }, [url, enabled])
   502→
   503→  useEffect(() => {
   504→    fetchData()
   505→  }, [fetchData])
   506→
   507→  return { data, loading, error, refetch: fetchData, pagination, stats }
   508→}
